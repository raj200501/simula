// actions.ts - candidate actions from the a11y tree, grouped LLM-Explorer style
// (one "abstract action" per repeated-sibling group), plus deterministic safety/scope rules.
import { UiNode, Rect, walkTree, labelOf } from "./uitree.js";

export type ActionKind = "tap" | "long_press" | "type" | "scroll" | "back";
export type Risk = "ok" | "destructive" | "purchase" | "external" | "out_of_scope";

export interface Candidate {
  key: string;             // abstract action id, stable across visits: stateHash + elementKey + kind
  elementKey: string;      // used to re-find the element on replay (never raw coordinates)
  kind: ActionKind;
  label: string;
  rect: Rect;
  groupSize: number;       // >1 => one of N similar items (feed card, chat, grid tile)
  risk: Risk;
  status: "untried" | "tried" | "ineffective" | "skipped" | "unreachable";
  priority: number;        // 0..3, set by the per-state LLM annotation (default 1)
  intent?: string;         // LLM label, e.g. "open character", "buy credits"
  repeats?: number;        // drain-probe counter (see Explorer.run)
}

// Deterministic guard rails. The LLM can ADD skips, never remove these.
const DESTRUCTIVE = /\b(log ?out|sign ?out|delete|remove account|deactivate|report|block|unsubscribe|clear (history|chat)|reset)\b/i;
const PURCHASE_CONFIRM = /\b(buy now|purchase|pay|confirm (purchase|payment)|subscribe now|start (free )?trial|1-tap buy)\b/i; // open the paywall, never confirm it
const EXTERNAL = /\b(share|rate us|rate the app|open in browser|privacy policy|terms( of (service|use))?|help center|contact us|instagram|tiktok|discord|twitter|x\.com)\b/i;
const OUT_OF_SCOPE = /\b(camera|take photo|gallery|photo library|upload (photo|image)|microphone|voice|record|attach|contacts|location|notifications? settings)\b/i;

export function classifyRisk(label: string): Risk {
  if (DESTRUCTIVE.test(label)) return "destructive";
  if (PURCHASE_CONFIRM.test(label)) return "purchase";
  if (OUT_OF_SCOPE.test(label)) return "out_of_scope";
  if (EXTERNAL.test(label)) return "external";
  return "ok";
}

/** Element key = path of (cls#rid) from the root with sibling indices only where needed + label with digits masked. */
export function elementKey(root: UiNode, target: UiNode): string {
  const path: string[] = [];
  const find = (n: UiNode): boolean => {
    if (n === target) return true;
    for (const c of n.children) {
      if (find(c)) {
        const same = n.children.filter((s) => s.cls === c.cls && s.rid === c.rid);
        path.unshift(`${c.cls}#${c.rid}${same.length > 1 ? `@${same.indexOf(c)}` : ""}`);
        return true;
      }
    }
    return false;
  };
  find(root);
  return `${path.join("/")}|${labelOf(target, 30).toLowerCase().replace(/\d+/g, "#")}`;
}

/** Re-resolve an element on the live tree during replay: exact key, else same label, else nearest same-class rect. */
export function resolveElement(root: UiNode, key: string, hint?: Rect): UiNode | undefined {
  const all = [...walkTree(root)].map((w) => w.node);
  const exact = all.find((n) => elementKey(root, n) === key);
  if (exact) return exact;
  const label = key.split("|")[1];
  const byLabel = label ? all.filter((n) => labelOf(n, 30).toLowerCase().replace(/\d+/g, "#") === label) : [];
  if (byLabel.length === 1) return byLabel[0];
  if (hint) {
    const cls = key.split("|")[0].split("/").pop()?.split("#")[0];
    return all.filter((n) => n.cls === cls).sort((a, b) => dist(a.rect, hint) - dist(b.rect, hint))[0];
  }
}
const dist = (a: Rect, b: Rect) => Math.hypot(a.x + a.w / 2 - b.x - b.w / 2, a.y + a.h / 2 - b.y - b.h / 2);

/**
 * Extract candidates. Rules:
 *  - actionable = clickable | longClickable | checkable | editable | scrollable (and enabled)
 *  - nested clickables: keep the INNERMOST clickable that has a label; a clickable container whose
 *    only clickable descendant covers >80% of it is dropped (Compose/RN double-wrapping)
 *  - repeated siblings (same shape) => one abstract action per group; `take` members get tried
 *  - scrollables get one "scroll" action (explorer caps scrolls per container)
 */
export function extractCandidates(root: UiNode, stateHash: string, take = 2): Candidate[] {
  const out: Candidate[] = [];
  const shape = (n: UiNode): string => `${n.cls}#${n.rid}(${n.children.map(shape).join(",")})`;
  // Repeated siblings. "Homogeneous" groups (feed cards, chat bubbles, grid tiles) get `take` members tried;
  // "heterogeneous" groups (bottom nav, tabs, chips: <=6 members, not in a scroller, all labels distinct)
  // are navigation, so every member is tried.
  const groupOf = new Map<UiNode, { size: number; idx: number; hetero: boolean }>();
  for (const { node } of walkTree(root)) {
    const shapes = node.children.map(shape);
    node.children.forEach((c, i) => {
      const members = node.children.filter((_, j) => shapes[j] === shapes[i]);
      if (members.length < 2) return;
      const labels = members.map((m) => labelOf(m, 30));
      const hetero = !node.scrollable && members.length <= 6 && labels.every((l) => l) && new Set(labels).size === labels.length;
      groupOf.set(c, { size: members.length, idx: members.indexOf(c), hetero });
    });
  }
  const inGroup = (n: UiNode, anc: UiNode[]): { size: number; idx: number; hetero: boolean } | undefined => {
    for (const a of [n, ...anc].reverse()) { const g = groupOf.get(a); if (g) return g; }
  };
  const visit = (n: UiNode, anc: UiNode[]) => {
    const act = n.enabled && (n.clickable || n.longClickable || n.checkable || n.editable);
    const innerClickable = [...walkTree(n)].slice(1).find((w) => w.node.clickable && area(w.node.rect) > 0.8 * area(n.rect));
    if (act && !innerClickable) {
      const g = inGroup(n, anc);
      if (!g || g.hetero || g.idx < take) {
        const label = labelOf(n) || n.rid || n.cls;
        const kind: ActionKind = n.editable ? "type" : n.clickable || n.checkable ? "tap" : "long_press";
        const ek = elementKey(root, n);
        out.push({ key: `${stateHash}:${ek}:${kind}`, elementKey: ek, kind, label, rect: n.rect,
          groupSize: g && !g.hetero ? g.size : 1, risk: classifyRisk(label), status: "untried", priority: 1 });
        if (n.longClickable && n.clickable) out.push({ ...out[out.length - 1], key: `${stateHash}:${ek}:long_press`, kind: "long_press" });
      }
    }
    if (n.scrollable && n.enabled) {
      const ek = elementKey(root, n);
      out.push({ key: `${stateHash}:${ek}:scroll`, elementKey: ek, kind: "scroll", label: `scroll ${n.rid || n.cls}`,
        rect: n.rect, groupSize: 1, risk: "ok", status: "untried", priority: 1 });
    }
    n.children.forEach((c) => visit(c, [...anc, n]));
  };
  visit(root, []);
  out.forEach((c) => { if (c.risk !== "ok") c.status = "skipped"; });
  return out;
}
const area = (r: Rect) => Math.max(0, r.w) * Math.max(0, r.h);

// ---------- leaving the app / out-of-scope surfaces (checked after EVERY action) ----------
export type Surface = "in_app" | "permission_dialog" | "camera" | "photo_picker" | "browser" | "play_billing"
  | "google_signin" | "os_settings" | "launcher" | "other_app";

export function classifyForeground(pkg: string, appPkg: string): Surface {
  if (pkg === appPkg) return "in_app";
  if (/permissioncontroller|packageinstaller/.test(pkg)) return "permission_dialog";
  if (/camera/i.test(pkg)) return "camera";
  if (/documentsui|photos|gallery|photopicker|mediaprovider/.test(pkg)) return "photo_picker";
  if (/chrome|browser|firefox|webview/.test(pkg)) return "browser";            // Custom Tabs show as chrome
  if (/com\.android\.vending/.test(pkg)) return "play_billing";                 // purchase sheet = monetization evidence!
  if (/com\.google\.android\.gms/.test(pkg)) return "google_signin";
  if (/com\.android\.settings/.test(pkg)) return "os_settings";
  if (/launcher|nexuslauncher/.test(pkg)) return "launcher";
  return "other_app";
}
